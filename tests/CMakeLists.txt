# Host-side unit tests for the MCORE network stack and inter-core primitives.
# Built with the NATIVE compiler, independently of the firmware cross-build:
#   cmake -S tests -B build-tests -G Ninja
#   cmake --build build-tests
#   ctest --test-dir build-tests --output-on-failure
cmake_minimum_required(VERSION 3.21)
project(mcore_tests C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(MCORE_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)

add_executable(net_tests
    test_main.cpp
    stubs.cpp
    test_eth_utils.cpp
    test_net.cpp
    test_tcp.cpp
    test_dhcp.cpp
    test_ipc.cpp
    test_client.cpp
    test_mqtt.cpp
    test_vda.cpp
    test_vda_ext.cpp
    test_sched.cpp
    # Real stack sources under test (system.cpp/eth.cpp are NOT linked —
    # their symbols are provided by stubs.cpp)
    ${MCORE_ROOT}/mcore/net/eth_utils.cpp
    ${MCORE_ROOT}/mcore/net/net.cpp
    ${MCORE_ROOT}/mcore/net/tcp.cpp
    ${MCORE_ROOT}/mcore/net/dhcp.cpp
    ${MCORE_ROOT}/mcore/net/arp.cpp
    ${MCORE_ROOT}/mcore/net/sntp.cpp
    ${MCORE_ROOT}/mcore/net/mqtt.cpp
    ${MCORE_ROOT}/mcore/vda5050/json.cpp
    ${MCORE_ROOT}/mcore/vda5050/order_manager.cpp
    ${MCORE_ROOT}/mcore/vda5050/messages.cpp
    ${MCORE_ROOT}/mcore/vda5050/geometry.cpp
    ${MCORE_ROOT}/mcore/vda5050/motion.cpp
    ${MCORE_ROOT}/mcore/vda5050/zones.cpp
    ${MCORE_ROOT}/mcore/sched/sched.cpp
    ${MCORE_ROOT}/mcore/sched/port_host.cpp
)

# Messages generated by test_vda.cpp, checked by tests/vda/validate.py
set(VDA_OUT_DIR ${CMAKE_CURRENT_BINARY_DIR}/vda_out)
file(MAKE_DIRECTORY ${VDA_OUT_DIR})
target_compile_definitions(net_tests PRIVATE MCORE_VDA_OUT_DIR="${VDA_OUT_DIR}" MCORE_SCHED_HOST)

target_include_directories(net_tests PRIVATE
    ${MCORE_ROOT}/mcore
    ${CMAKE_CURRENT_SOURCE_DIR}
)

if(NOT MSVC)
    target_compile_options(net_tests PRIVATE -Wall -Wextra)
endif()

# test_ipc.cpp runs producer/consumer on two threads
find_package(Threads REQUIRED)
target_link_libraries(net_tests PRIVATE Threads::Threads)

# Self-contained exe on Windows: avoid libstdc++/libgcc/winpthread DLL deps
if(WIN32 AND NOT MSVC)
    target_link_options(net_tests PRIVATE -static)
endif()

# ---------------------------------------------------------------------------
# TLS: Mbed TLS (third_party, MCORE configuration) built for the host, plus
# the server side for the in-memory handshake tests.
# ---------------------------------------------------------------------------
file(GLOB MBEDTLS_SRC ${MCORE_ROOT}/third_party/mbedtls/library/*.c)
add_library(mbedtls_host STATIC ${MBEDTLS_SRC})
target_include_directories(mbedtls_host PUBLIC
    ${MCORE_ROOT}/third_party/mbedtls/include
    ${MCORE_ROOT}/mcore/net/tls)
target_compile_definitions(mbedtls_host PUBLIC
    MBEDTLS_CONFIG_FILE="mbedtls_config_mcore.h"
    MCORE_TLS_TEST_SERVER
    MCORE_TLS_MEMORY_DEBUG)
if(WIN32)
    target_link_libraries(mbedtls_host PUBLIC ws2_32)   # inet_pton in x509_crt.c
endif()

add_executable(tls_tests
    test_tls.cpp
    ${MCORE_ROOT}/mcore/net/tls/tls_client.cpp
    ${MCORE_ROOT}/mcore/net/tls/tls_platform.cpp
)
target_include_directories(tls_tests PRIVATE ${MCORE_ROOT}/mcore ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(tls_tests PRIVATE
    MCORE_TEST_CERTS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/certs")
target_link_libraries(tls_tests PRIVATE mbedtls_host)
if(NOT MSVC)
    target_compile_options(tls_tests PRIVATE -Wall -Wextra)
endif()
if(WIN32 AND NOT MSVC)
    target_link_options(tls_tests PRIVATE -static)
endif()

# Live check against a running broker (tests/broker/run_broker.sh).
# Built always; registered with CTest only with -DMCORE_LIVE_TESTS=ON.
add_executable(mqtt_live
    live/mqtt_live.cpp
    ${MCORE_ROOT}/mcore/net/mqtt.cpp
    ${MCORE_ROOT}/mcore/net/tls/tls_client.cpp
    ${MCORE_ROOT}/mcore/net/tls/tls_platform.cpp
)
target_include_directories(mqtt_live PRIVATE ${MCORE_ROOT}/mcore)
target_compile_definitions(mqtt_live PRIVATE
    MCORE_LIVE_CA_FILE="${MCORE_ROOT}/tests/broker/pki/ca.crt")
target_link_libraries(mqtt_live PRIVATE mbedtls_host)
if(WIN32 AND NOT MSVC)
    target_link_options(mqtt_live PRIVATE -static)
endif()

enable_testing()
add_test(NAME net_tests COMMAND net_tests)
add_test(NAME tls_tests COMMAND tls_tests)

# VDA 5050 messages vs the official JSON schemas (needs: pip install jsonschema)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME vda_schemas
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/vda/validate.py ${VDA_OUT_DIR})
    set_tests_properties(vda_schemas PROPERTIES DEPENDS net_tests)
    # Register headers vs the vendor CMSIS device headers (base addresses,
    # register offsets, bit fields): catches SVD errors and generator bugs
    set(REG_CHECK ${CMAKE_CURRENT_SOURCE_DIR}/../codegen/registers/check_regs.py)
    foreach(chip IN ITEMS stm32h755 stm32f767)
        add_test(NAME regs_${chip}
            COMMAND ${Python3_EXECUTABLE} -I ${REG_CHECK}
                    --regs ${CMAKE_CURRENT_SOURCE_DIR}/../mcore/chip/${chip}/regs.hpp
                    --cmsis ${CMAKE_CURRENT_SOURCE_DIR}/../codegen/nvic/${chip}xx.h)
    endforeach()
endif()
option(MCORE_LIVE_TESTS "Run mqtt_live against a local broker" OFF)
if(MCORE_LIVE_TESTS)
    add_test(NAME mqtt_live COMMAND mqtt_live)
endif()
